var results = new System.Collections.Generic.List<object>();
var outputs = new System.Collections.Generic.Dictionary<string, UnityEngine.Vector4[]>();
foreach (string variant in new[] { "Current", "Baseline" })
{
    var compute = UnityEngine.Object.Instantiate(UnityEditor.AssetDatabase.LoadAssetAtPath<UnityEngine.ComputeShader>("Assets/Tests/GrassAuditProbe/" + variant + ".compute"));
    var ids = new UnityEngine.Texture2D(2,2,UnityEngine.TextureFormat.RGBAFloat,false,true);
    var weights = new UnityEngine.Texture2D(2,2,UnityEngine.TextureFormat.RGBAFloat,false,true);
    var parameters = new UnityEngine.ComputeBuffer(2,80);
    var output = new UnityEngine.ComputeBuffer(8,16);
    try
    {
        parameters.SetData(new UnityEngine.Vector4[] {
            new(1,.8f,.04f,.6f),new(40,5,.2f,1),new(.2f,.8f,.4f,1),new(.7f,.8f,.9f,1),new(.9f,1,.8f,1),
            new(.25f,.4f,.02f,.2f),new(20,3,.4f,1),new(.6f,.4f,.2f,1),new(.8f,.7f,.6f,1),new(.6f,.8f,1,1)
        });
        int kernel=compute.FindKernel("Probe");
        compute.SetBuffer(kernel,"_Params",parameters);
        compute.SetBuffer(kernel,"_Result",output);
        compute.SetTexture(kernel,"_BiomeIds",ids);
        compute.SetTexture(kernel,"_BiomeWeights",weights);
        compute.SetVector("_BiomeMap_TexelSize",new(.5f,.5f,2,2));
        foreach (string scenario in new[] { "packed", "split", "unequal", "empty", "small" })
        {
            var idPixels=new UnityEngine.Color[4];
            var weightPixels=new UnityEngine.Color[4];
            for(int i=0;i<4;i++)
            {
                bool right=(i%2)==1;
                idPixels[i]=new UnityEngine.Color(scenario=="unequal" && right ? 1f/255f : 0,0,0,0);
                float weight=scenario=="packed" ?.5f:scenario=="empty" ?0:((scenario=="split" || scenario=="small") && right ?0:1);
                weightPixels[i]=new UnityEngine.Color(weight,0,0,0);
            }
            ids.SetPixels(idPixels);ids.Apply();weights.SetPixels(weightPixels);weights.Apply();
            compute.SetVector("_ProbeUv",new(scenario=="small" ?.749975f:.5f,.5f,0,0));
            compute.Dispatch(kernel,1,1,1);
            var values=new UnityEngine.Vector4[8];output.GetData(values);
            outputs[variant+"/"+scenario]=values;
            results.Add(new {variant,scenario,values=System.Array.ConvertAll(values,v=>new float[]{v.x,v.y,v.z,v.w})});
        }
    }
    finally
    {
        parameters.Dispose();output.Dispose();
        UnityEngine.Object.DestroyImmediate(ids);UnityEngine.Object.DestroyImmediate(weights);UnityEngine.Object.DestroyImmediate(compute);
    }
}
